cmake_minimum_required(VERSION 3.20)
project(swerve_local_tests LANGUAGES CXX)
include(CTest)
if(NOT EXISTS "${SWERVE_CHASSIS_SOURCE_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Set SWERVE_CHASSIS_SOURCE_DIR to swerve_chassis_core sources")
endif()
get_filename_component(MPPI_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_subdirectory("${SWERVE_CHASSIS_SOURCE_DIR}" chassis)
add_subdirectory("${MPPI_ROOT}/swerve_mppi_core" core)
add_executable(record_portable "${MPPI_ROOT}/swerve_mppi_ros/tests/record_portable.cpp")
target_compile_definitions(record_portable PRIVATE SWERVE_HEADING_REFERENCE=1)
target_link_libraries(record_portable PRIVATE swerve_mppi::core)
target_include_directories(record_portable PRIVATE "${MPPI_ROOT}/swerve_mppi_ros/include")
target_compile_options(record_portable PRIVATE -Wall -Wextra -Wpedantic -Werror)
foreach(name support_tests closed_loop stagnation)
  add_executable(${name} "${MPPI_ROOT}/swerve_mppi_ros/tests/${name}.cpp")
  target_link_libraries(${name} PRIVATE swerve_mppi::core)
  target_include_directories(${name} PRIVATE "${MPPI_ROOT}/swerve_mppi_ros/include")
  target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME ${name} COMMAND ${name})
endforeach()
set_tests_properties(closed_loop stagnation PROPERTIES TIMEOUT 240)

add_test(NAME installed_consumer COMMAND ${CMAKE_COMMAND}
  -DBUILD_DIR=${CMAKE_BINARY_DIR} -DSOURCE_DIR=${MPPI_ROOT}/swerve_mppi_core
  -P ${MPPI_ROOT}/swerve_mppi_core/tests/install_test.cmake)
