cmake_minimum_required(VERSION 3.20)
project(swerve_mppi_ros VERSION 0.29.0 LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(swerve_mppi_core 0.28 CONFIG REQUIRED)
set(dependencies rclcpp rclcpp_lifecycle std_msgs geometry_msgs nav_msgs diagnostic_msgs swerve_gazebo_sim tf2_ros tf2_geometry_msgs nav2_core nav2_costmap_2d pluginlib)
foreach(dep ${dependencies})
  find_package(${dep} REQUIRED)
endforeach()
add_library(swerve_mppi_nav2 SHARED src/nav2_controller.cpp src/ordered_goal_checker.cpp)
add_library(swerve_mppi_chassis_bridge SHARED src/chassis_bridge.cpp)
foreach(target swerve_mppi_nav2 swerve_mppi_chassis_bridge)
  target_compile_features(${target} PUBLIC cxx_std_17)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  target_include_directories(${target} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
  target_link_libraries(${target} PUBLIC swerve_mppi::core)
  ament_target_dependencies(${target} PUBLIC ${dependencies})
endforeach()
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
add_executable(chassis_bridge src/main.cpp)
target_link_libraries(chassis_bridge PRIVATE swerve_mppi_chassis_bridge)
install(TARGETS swerve_mppi_nav2 swerve_mppi_chassis_bridge EXPORT export_ros ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(TARGETS chassis_bridge RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_export_targets(export_ros HAS_LIBRARY_TARGET)
ament_export_dependencies(swerve_mppi_core ${dependencies})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(ros_tests tests/ros_tests.cpp TIMEOUT 120)
  target_link_libraries(ros_tests swerve_mppi_nav2 swerve_mppi_chassis_bridge)
  set_tests_properties(ros_tests PROPERTIES ENVIRONMENT "ROS_DOMAIN_ID=83;ROS_AUTOMATIC_DISCOVERY_RANGE=LOCALHOST")
endif()
ament_package()
