cmake_minimum_required(VERSION 3.20)
project(swerve_support_tests LANGUAGES CXX)
find_package(swerve_mppi_core 0.28 CONFIG REQUIRED)
enable_testing()
foreach(name support_tests chassis_driver)
 add_executable(${name} ../${name}.cpp)
 target_link_libraries(${name} PRIVATE swerve_mppi::core)
 target_include_directories(${name} PRIVATE ../../include)
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
add_test(NAME caller_support COMMAND support_tests)
set(CHASSIS_SOURCE_DIR "" CACHE PATH "Optional current Python chassis checkout")
if(CHASSIS_SOURCE_DIR)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_test(NAME python_chassis COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../python_chassis_test.py --tool $<TARGET_FILE:chassis_driver> --chassis-root ${CHASSIS_SOURCE_DIR})
 set_tests_properties(python_chassis PROPERTIES TIMEOUT 180)
endif()
