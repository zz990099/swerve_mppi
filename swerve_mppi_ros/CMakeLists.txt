cmake_minimum_required(VERSION 3.20)
project(swerve_mppi_ros VERSION 0.28.0 LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(swerve_mppi_core 0.28 CONFIG REQUIRED)
foreach(dep rclcpp std_msgs geometry_msgs nav_msgs diagnostic_msgs swerve_gazebo_sim tf2_ros tf2_geometry_msgs)
 find_package(${dep} REQUIRED)
endforeach()
add_library(swerve_mppi_ros_node SHARED src/controller_node.cpp)
target_compile_features(swerve_mppi_ros_node PUBLIC cxx_std_17)
target_compile_options(swerve_mppi_ros_node PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(swerve_mppi_ros_node PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(swerve_mppi_ros_node PUBLIC swerve_mppi::core)
ament_target_dependencies(swerve_mppi_ros_node PUBLIC rclcpp std_msgs geometry_msgs nav_msgs diagnostic_msgs swerve_gazebo_sim tf2_ros tf2_geometry_msgs)
add_executable(controller src/main.cpp)
target_link_libraries(controller PRIVATE swerve_mppi_ros_node)
install(TARGETS swerve_mppi_ros_node EXPORT export_node ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(TARGETS controller RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_export_targets(export_node HAS_LIBRARY_TARGET)
ament_export_dependencies(swerve_mppi_core rclcpp std_msgs geometry_msgs nav_msgs diagnostic_msgs swerve_gazebo_sim tf2_ros tf2_geometry_msgs)
if(BUILD_TESTING)
 find_package(ament_cmake_gtest REQUIRED)
 ament_add_gtest(ros_tests tests/ros_tests.cpp TIMEOUT 120)
 target_link_libraries(ros_tests swerve_mppi_ros_node)
 set_tests_properties(ros_tests PROPERTIES ENVIRONMENT "ROS_DOMAIN_ID=83;ROS_AUTOMATIC_DISCOVERY_RANGE=LOCALHOST")
endif()
ament_package()
