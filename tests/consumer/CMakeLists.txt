cmake_minimum_required(VERSION 3.20)
project(swerve_mppi_consumer LANGUAGES CXX)
find_package(swerve_mppi 0.25 CONFIG REQUIRED)
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE swerve_mppi::core)
# Compile every installed public header in its own translation unit. Moving
# modules must not hide missing includes behind another consumer header.
get_target_property(core_include_dirs swerve_mppi::core INTERFACE_INCLUDE_DIRECTORIES)
set(public_headers)
foreach(include_dir IN LISTS core_include_dirs)
  file(GLOB_RECURSE installed_headers CONFIGURE_DEPENDS
    "${include_dir}/swerve_mppi/*.hpp")
  list(APPEND public_headers ${installed_headers})
endforeach()
list(REMOVE_DUPLICATES public_headers)
if(NOT public_headers)
  message(FATAL_ERROR "No installed swerve_mppi public headers found")
endif()
foreach(header IN LISTS public_headers)
  string(MAKE_C_IDENTIFIER "${header}" header_id)
  set(probe "${CMAKE_CURRENT_BINARY_DIR}/headers/${header_id}.cpp")
  file(WRITE "${probe}" "#include \"${header}\"\n")
  target_sources(consumer PRIVATE "${probe}")
endforeach()
enable_testing()
add_test(NAME consumer_runs COMMAND consumer)
